ALTER TABLE ApplicantData_Cleaned
ADD CONSTRAINT pk_applicant PRIMARY KEY (App_ID);


ALTER TABLE CampaignData_Cleaned
ADD CONSTRAINT pk_campaign PRIMARY KEY (ID);


CREATE OR REPLACE VIEW master_applicant_campaign_outreach AS
SELECT
    a.App_ID,
    a.Country,
    a.University AS applicant_university,
    a.Phone_number,
    o.Reference_ID,
    o.Received_at,
    o.University AS outreach_university,
    o.Caller_Name,
    o.Outcome_1,
    o.Remark,
	o.Campaign_ID,
    o.Escalation_Required,
    c.ID AS original_campaign_id,
    c.Name AS campaign_name,
    c.Category,
    c.Intake,
    c.University AS campaign_university,
    c.Status,
    c.Start_Date
FROM ApplicantData_Cleaned a
LEFT JOIN OutreachData_Cleaned o
    ON a.App_ID = o.Reference_ID
LEFT JOIN CampaignData_Cleaned c
    ON o.Campaign_ID = c.ID;


-- Index on OutreachData.Campaign_ID
CREATE INDEX idx_outreach_campaign_id
    ON OutreachData_Cleaned(Campaign_ID);

-- Index on OutreachData.Reference_ID
CREATE INDEX idx_outreach_reference_id
    ON OutreachData_Cleaned(Reference_ID);


SELECT * FROM master_applicant_campaign_outreach LIMIT 20;
